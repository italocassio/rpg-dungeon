## Luz de personagem: emissão igual ao albedo, energia .20. Peças de sombra ficam de fora.
static func light_up(rig: Node, seen: Dictionary = {}) -> void:
	for child in rig.get_children():
		var m := child as MeshInstance3D
		if m != null and m.material_override is StandardMaterial3D:
			var src: StandardMaterial3D = m.material_override
			var key: String = src.albedo_color.to_html(false).to_upper()
			if not src.emission_enabled and key not in ["1A1410", "2A1E18", "3A2E25", "14231F"]:
				if not seen.has(src):
					var dup: StandardMaterial3D = src.duplicate()
					dup.emission_enabled = true
					dup.emission = src.albedo_color
					dup.emission_energy_multiplier = 0.2
					seen[src] = dup
				m.material_override = seen[src]
		light_up(child, seen)


static func dbl(c: Color, r := 0.85) -> Material:
	var m: StandardMaterial3D = Look.matte(c, r).duplicate()
	m.cull_mode = BaseMaterial3D.CULL_DISABLED
	return m


static func mirror(p: PackedVector2Array) -> PackedVector2Array:
	var o := PackedVector2Array()
	for i in range(p.size() - 1, -1, -1):
		o.append(Vector2(-p[i].x, p[i].y))
	return o


static func seal(rig: CreatureRig, parent: Node3D, pos: Vector3, color: Color, energy: float, back := false) -> void:
